\section{Tools}

\begin{frame}{What is a tool? An example first: \texttt{read\_file}}
  \small
  \texttt{read\_file} is a function $f$ that takes a path (given as JSON) as input and returns a string:
  \[
    f : \textcolor{actc}{\texttt{\{"path": string\}}} \longrightarrow \textcolor{obsc}{\texttt{string}}
  \]
  \vspace{-4pt}
  \[
    f\big(\textcolor{actc}{\texttt{\{"path": "stats.py"\}}}\big) = \textcolor{obsc}{\texttt{"def median(xs): \ldots"}}
  \]

  \onslide<2->{%
  \centering
  \begin{tikzpicture}
    \node[draw=greyc,rounded corners,thick,inner sep=8pt] (d) {\blurred{def read\_file(path):\\
\ \ \ \ with open(path) as fh:\\
\ \ \ \ \ \ \ \ return fh.read()}};
    \node[anchor=west,text width=4.2cm,align=left,font=\footnotesize,text=greyc] at ($(d.east)+(0.4,0)$)
      {The LLM does not see (and does not care to see) how $f$ is defined.};
  \end{tikzpicture}}
\end{frame}

\begin{frame}{What is a tool?}
  A tool is an ``interface'' to a function, i.e., 
  \begin{small}
  \[
    \tau = \Big(\ \underbrace{\textcolor{actc}{\text{name},\ S}}_{\textcolor{actc}{\text{interface --- sent to the model}}}\ ,\ \ \underbrace{\textcolor{greyc}{f}}_{\textcolor{greyc}{\text{implementation --- host-only}}}\ \Big)
  \]
  \end{small}
  \vspace{-6pt}
  \begin{footnotesize}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item The \textcolor{actc}{interface} $(\text{name}, S)$ is what the model reads and calls against. The tool has a name and $S$ is the \emph{schema} (usually a JSON Schema). 
    \item The \textcolor{greyc}{implementation} $f : S \to \mathcal{Y}$ is what runs, on the host, and is never sent to the model.
    \item $\mathcal{X}$: all possible arguments ($S \subseteq \mathcal{X}$, the valid ones). $\mathcal{Y}$: all possible results.
  \end{itemize}
  \end{footnotesize}
  \vfill
  \footnotesize The model writes down a call from the interface alone. The host checks $x \in S$, then runs $f$. If $x \notin S$, it returns an error message instead.
\end{frame}

\begin{frame}{Let's fill it in: $\tau = \texttt{edit\_file}$}
  \small
  \textbf{In plain English:} ``\texttt{edit\_file(path, old\_text, new\_text)} replaces \texttt{old\_text} with \texttt{new\_text} in the file at \texttt{path}. If it can't, it says why.''
  \vspace{4pt}

  \onslide<1->{\footnotesize\color{greyc} What would you put in each blank? Call it out before the click.}
  \vspace{6pt}

  \begin{tabular}{@{}l P{9.6cm}@{}}
    $S$ & \redact{2-}{objects with exactly the keys \texttt{path}, \texttt{old\_text}, \texttt{new\_text}} \\[6pt]
    $f$ & \redact{3-}{replace \texttt{old\_text} by \texttt{new\_text} in the file; return ``ok'' or an error message} \\[6pt]
    $\mathcal{Y}$ & \redact{4-}{string} \\
  \end{tabular}
\end{frame}

\begin{frame}{Tools in the example}
  \small
  \begin{tabular}{@{}l P{3.6cm} P{3.4cm} P{3.4cm}@{}}
    \textbf{name} & \textbf{$S$: allowed arguments} & \textbf{$f$: what the host does} & \textbf{result in $\mathcal{Y}$} \\ \hline
    \rule{0pt}{12pt}\texttt{read\_file} & a path (string) & reads the file & its contents (a string) \\[3pt]
    \texttt{run\_tests} & none, or a path & runs the test suite & pass/fail summary, traceback \\[3pt]
    \texttt{edit\_file} & a path, the old text, the new text & replaces old with new & ``ok'' or an error message \\
  \end{tabular}
\end{frame}

\begin{frame}{What do I know?}
  \small What the model receives on its next turn:\\[4pt]
  \centering
  \ctx{
    \crow{system}{You are a coding agent working in this repository. \ldots}
    \crow{tools}{read\_file(path) --- Read a file and return its contents.\newline
                 run\_tests(path?) --- Run the test suite; return a summary.\newline
                 edit\_file(path, old\_text, new\_text) --- Replace old\_text with new\_text.}
    \crow{user}{The test suite is failing. Fix it.}
    \crow{\textcolor{actc}{model}}{\textcolor{actc}{call(run\_tests, \{\})}}
    \crow{\textcolor{obsc}{result}}{\textcolor{obsc}{FAILED test\_median\_even: assert 3 == 2.5. 1 failed, 2 passed}}
    \crow{\textcolor{actc}{model}}{\textcolor{actc}{call(read\_file, \{path: "stats.py"\})}}
    \crow{\textcolor{obsc}{result}}{\textcolor{obsc}{def median(xs): xs = sorted(xs); n = len(xs); return xs[n // 2] \ldots}}
    \crow{\textcolor{actc}{model}}{\textcolor{greyc}{\rmfamily\itshape $\leftarrow$ writes its next action here}}
  }

  \raggedright\footnotesize
  \vspace{4pt}
  Each model turn is either a \textcolor{actc}{\texttt{call}} (run a tool, keep going) or an \textcolor{actc}{\texttt{answer}} (reply to the user, stop).
\end{frame}

\begin{frame}{What do I know?}
  \textbf{Context:} a history of instructions, actions, and results. It starts as
  \[
    h_0 = \big(\text{system prompt},\ \{(\text{name}_\tau, S_\tau)\}_{\tau \in T},\ \text{user request}\big)
  \]
  and each step appends one (action, result) pair:
  \[
    h_{t+1} = h_t \frownop (\textcolor{actc}{a_t}, \textcolor{obsc}{o_t}),
    \qquad
    h_t = h_0 \frownop (\textcolor{actc}{a_0}, \textcolor{obsc}{o_0}) \frownop \cdots \frownop (\textcolor{actc}{a_{t-1}}, \textcolor{obsc}{o_{t-1}}) \in H
  \]
  \footnotesize\raggedright
  \textcolor{actc}{$a_t$}: the model's action at step $t$. \textcolor{obsc}{$o_t$}: the tool's output for that action. $\frownop$ appends to the end. $H$: all possible histories.
  \vfill
  \small
  \onslide<2->{\textbf{Conditioning:} the model's next choice depends on $h_t$ and nothing else. If it isn't in $h_t$, the model doesn't know it.}
\end{frame}

\begin{frame}{What do I do?}
  \footnotesize
  \textbf{The only options:} call a tool (\texttt{read\_file}, \texttt{run\_tests}, \texttt{edit\_file}, \textcolor{greyc}{\ldots}) \textbf{or} answer the user.\\[8pt]
  \centering
  {\footnotesize Probability of each next action, given the context $h_2$ from before}\\[4pt]
  \begin{tikzpicture}[x=1cm,y=-0.55cm]
    \foreach \i/\lab/\pr in {
        0/{call(edit\_file, \{old\_text: "return xs[n // 2]", \ldots\})}/0.55,
        1/{call(edit\_file, \{old\_text: "return xs[n//2]", \ldots\})}/0.25,
        2/{call(run\_tests, \{\})}/0.10,
        3/{call(read\_file, \{path: "stats.py"\})}/0.06,
        4/{answer("Found the bug in median.")}/0.03,
        5/{other calls to these tools or other answers}/0.01} {
      \node[anchor=east,font=\ttfamily\scriptsize] at (0,\i) {\lab};
      \fill[actc] (0.15,\i-0.2) rectangle ({0.15+5*\pr},\i+0.2);
      \node[anchor=west,font=\footnotesize] at ({0.25+5*\pr},\i) {\pr};
    }
    \draw[visible on=<2->,hilc,thick,rounded corners=3pt] (-8.3,-0.45) rectangle (3.9,1.45);
  \end{tikzpicture}

  \raggedright\footnotesize
  \onslide<2->{\textcolor{hilc}{Same tool, different arguments: different actions.} Choosing the tool and writing its arguments is one choice.}
\end{frame}

\begin{frame}{Putting it together}
  \begin{columns}[c]
    \begin{column}{0.45\textwidth}
      \centering
      \begin{tikzpicture}[
          box/.style={draw,rounded corners,thick,minimum width=2.9cm,minimum height=0.8cm,align=center,font=\footnotesize},
          lab/.style={font=\scriptsize},
          >=Stealth]
        \node[box,draw=greyc] (h) {context $h_t$};
        \node[box,draw=actc,below=0.9cm of h] (p) {LLM: $a_t \sim \pi(\cdot \mid h_t)$};
        \node[box,draw=obsc,below=0.9cm of p,visible on=<3->] (o) {host: $o_t = o(\tau, x)$};
        \node[box,draw=hilc,minimum width=1.1cm,right=1.1cm of p,visible on=<2->] (s) {stop};
        \draw[->,thick] (h) -- (p);
        \draw[->,thick,actc,visible on=<2->] (p) -- node[above,lab] {answer} (s);
        \draw[->,thick,actc,visible on=<3->] (p) -- node[right,lab] {call$(\tau, x)$} (o);
        \draw[->,thick,obsc,visible on=<4->] (o.west) -- ++(-0.5,0) coordinate (c)
          -- node[midway,rotate=90,above,lab] {append $(a_t, o_t)$} (c |- h.west) -- (h.west);
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.55\textwidth}
      \footnotesize
      \begin{enumerate}
        \item<1-> Sample an action: $a_t \sim \pi(\cdot \mid h_t)$.
        \item<2-> If $a_t = \text{answer}(m)$: stop and return $m$.
        \item<3-> If $a_t = \text{call}(\tau, x)$: the host returns
          \[ o_t = \begin{cases} f_\tau(x) & x \in S_\tau \\ \text{an error message} & \text{otherwise} \end{cases} \]
        \item<4-> Append: $h_{t+1} = h_t \frownop (a_t, o_t)$. Repeat.
        \item<5-> Also stop when $t = N$, a step budget.
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The loop, on our example}
  \small
  $h_0$ = ``The test suite is failing. Fix it.'' Each step appends one (action, result) pair. What does each step return?\\[4pt]
  \footnotesize
  \setlength{\tabcolsep}{4pt}\begin{tabular}{@{}c l l@{}}
    $t$ & \textcolor{actc}{action $a_t$} & \textcolor{obsc}{result $o_t$} \\ \hline
    \rule{0pt}{11pt}0 & \texttt{call(run\_tests, \{\})} & 1 failed, 2 passed \\[2pt]
    1 & \texttt{call(read\_file, \{path: "stats.py"\})} & the source of \texttt{stats.py} \\[2pt]
    2 & \texttt{call(edit\_file, \{old\_text: "return xs[n//2]", \ldots\})} & \textcolor{hilc}{\redact{2-}{error: old\_text not found}} \\[2pt]
    \visible<3->{3} & \visible<3->{\texttt{call(edit\_file, \{old\_text: "return xs[n // 2]", \ldots\})}} & \visible<3->{ok} \\[2pt]
    \visible<3->{4} & \visible<3->{\texttt{call(run\_tests, \{\})}} & \visible<3->{3 passed} \\[2pt]
    \visible<3->{5} & \visible<3->{\texttt{answer("Fixed median; 3 tests pass.")}} & \visible<3->{\redact{4-}{(loop stops)}} \\
  \end{tabular}\\[4pt]
  \only<1>{\footnotesize\color{greyc} Step 2's old\_text looks right -- what actually happens?}%
  \only<2>{\footnotesize\color{greyc} $o_2$ is an error (the file has \texttt{xs[n // 2]}); it lands in $h_3$, and the loop continues.}%
  \only<3>{\footnotesize\color{greyc} Step 5 is an answer. Does the loop take another step?}%
  \only<4>{\footnotesize\color{greyc} $a_5 = \text{answer}(m)$, so the loop stops and returns $m$. There is no $o_5$ and no $h_6$.}%
\end{frame}

\begin{frame}{Consequences}
  \small
  \begin{itemize}\setlength{\itemsep}{10pt}
    \item<1-> \textbf{The context only grows.} Each step appends to $h_t$, but the model can only read a bounded amount. \textbf{Compaction} keeps $h_0$ and the last $k$ steps, and replaces the rest by a summary $s$ written by the LLM:
    \[
      h_t \ \longmapsto\ h_0 \frownop s \frownop (a_{t-k}, o_{t-k}) \frownop \cdots \frownop (a_{t-1}, o_{t-1})
    \]
    \item<2-> \textbf{Tools are simply side-effectful functions.} $f_\tau$ can do anything that ends in a result, including running another loop. A \emph{sub-agent} is one tool call, e.g.\ \texttt{call(delegate, \{task: "find every caller of median"\})}:
    \[
      f_{\texttt{delegate}}(x) = m \quad \text{where} \quad \underbrace{h_0' = (\ldots,\ x)}_{\text{a fresh loop}} \ \rightsquigarrow\ \text{answer}(m)
    \]
    {\color{greyc} How do sub-agents affect the context?}
  \end{itemize}
\end{frame}

    \item An agent is a model in a loop: it emits structured calls, the host validates and runs them, and the results go back into the history.
    \item Reliability is a design problem: precise schemas, small scoped tool sets, and verification steps (like rerunning the tests) make errors easier to catch.
  \end{enumerate}
\end{frame}
